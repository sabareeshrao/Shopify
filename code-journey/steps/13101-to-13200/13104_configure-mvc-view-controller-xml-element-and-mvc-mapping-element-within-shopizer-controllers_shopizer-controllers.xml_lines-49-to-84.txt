 		<mvc:interceptor>

  			<bean class="com.salesmanager.shop.filter.CorsFilter"/>
 		</mvc:interceptor>
 		
 		<!-- customer section filter -->
 		<mvc:interceptor>
  			<mvc:mapping path="/customer/**"/>

 		</mvc:interceptor>
 		
 		 <!-- REST api -->
 		<mvc:interceptor>
