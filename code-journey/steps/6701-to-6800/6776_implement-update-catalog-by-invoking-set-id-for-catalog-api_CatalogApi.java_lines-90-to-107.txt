
  @PatchMapping(value = "/private/catalog/{id}")
  @ResponseStatus(HttpStatus.OK)
  @ApiOperation(httpMethod = "PATCH", value = "Update catalog", notes = "",
      response = Void.class)

      @ApiImplicitParam(name = "store", dataType = "String", defaultValue = "DEFAULT"),
