				if (newItemValue.getQuantity() == 0) {
					// remove from cart
					shoppingCartService.deleteShoppingCartItem(oldCartItem.getId());
					cartModel.getLineItems().remove(oldCartItem);
					++itemUpdatedCnt;
					continue;
				}
				// update qty
				oldCartItem.setQuantity(newItemValue.getQuantity());
				++itemUpdatedCnt;
