	public void contextInitialized(ServletContextEvent servletContextEvent) {
		logger.info("===context init===");
		System.getenv().forEach((k, v) -> {
		  logger.debug(k + ":" + v);
		});
		Properties props = System.getProperties();
		props.forEach((k, v) -> {
		  logger.debug(k + ":" + v);
		});
	}
