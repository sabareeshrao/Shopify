	        //may affect the sub total

	        if(CollectionUtils.isNotEmpty(orderTotalVariation.getVariations())) {

	        		variation.setSortOrder(currentCount++);
	        		orderTotals.add(variation);
	        		subTotal = subTotal.subtract(variation.getValue());
	        	}
