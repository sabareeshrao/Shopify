	      + "left join fetch p.merchantStore pm "
	      + "left join fetch p.prices pp "
	      + "left join fetch pp.descriptions ppd "
	      + "join fetch p.product ppr "
	      + "left join fetch ppr.descriptions pprd "
	      + "left join fetch p.productVariant ppi "
